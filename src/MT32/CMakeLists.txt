include(FetchContent)
set(MECHWARRIOR_MT32_SOURCE_DIR "" CACHE PATH "Existing Munt checkout (optional)")
if(MECHWARRIOR_MT32_SOURCE_DIR)
    set(munt_SOURCE_DIR "${MECHWARRIOR_MT32_SOURCE_DIR}")
else()
    FetchContent_Declare(munt
        GIT_REPOSITORY https://github.com/munt/munt.git
        GIT_TAG 6e7c01fba7e1d50c8fa705834889fd0eac136075)
    FetchContent_GetProperties(munt)
    if(NOT munt_POPULATED)
        FetchContent_Populate(munt)
    endif()
endif()
enable_language(CXX)
set(munt_SOURCE_DIR "${munt_SOURCE_DIR}" PARENT_SCOPE)
set(libmt32emu_SHARED ON CACHE BOOL "" FORCE)
set(libmt32emu_C_INTERFACE ON CACHE BOOL "" FORCE)
set(libmt32emu_CPP_INTERFACE OFF CACHE BOOL "" FORCE)
set(libmt32emu_BUILD_TESTING OFF CACHE BOOL "" FORCE)
add_subdirectory("${munt_SOURCE_DIR}/mt32emu" "${CMAKE_CURRENT_BINARY_DIR}/libmt32emu")
add_library(mechwarrior_mt32 STATIC MT32.c)
target_include_directories(mechwarrior_mt32 PUBLIC ../include)
target_link_libraries(mechwarrior_mt32 PUBLIC MT32Emu::mt32emu)
